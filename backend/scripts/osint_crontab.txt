# FACE.ai — Planification des ingestions OSINT (v030)
# Données OPEN SOURCE, personnes publiques du corpus uniquement. Cf. CLAUDE.md.
#
# ⚠ MÉCANISME INSTALLÉ = launchd (pas cron). Les agents sont versionnés dans
#   scripts/launchd/ai.face.osint.*.plist et chargés dans ~/Library/LaunchAgents/.
#   launchd relance au réveil si la machine dormait et évite le souci de Full
#   Disk Access de /usr/sbin/cron. Wrapper : scripts/osint-run.sh.
#   (Re)charger : launchctl load -w ~/Library/LaunchAgents/ai.face.osint.*.plist
#   Décharger   : launchctl unload   ~/Library/LaunchAgents/ai.face.osint.*.plist
#   Lancer à la demande : launchctl kickstart gui/$(id -u)/ai.face.osint.<task>
#   Logs : /tmp/faceai-osint-launchd.log + backend/logs/ingest_{source}_{date}.log
#
# Les lignes crontab ci-dessous restent une ALTERNATIVE (si tu préfères cron).
# Toutes les commandes tournent dans le conteneur `api` (qui a database/config).

REPO=/Users/patrickostertag/Documents/DataForIA/FACE.ai

# ── Pays (transversal) : Wikidata P27→P297. Mensuel, après enrichissement.
0 4 1 * *   cd $REPO && docker compose exec -T api python scripts/enrich_entity_countries.py

# ── P1A OpenSanctions (PEP/sanctions) — hebdomadaire (bulk download).
0 2 * * 1   cd $REPO && docker compose exec -T api python scripts/ingest_opensanctions.py

# ── P1B Wikimedia Commons (portraits libres) — hebdomadaire.
0 5 * * 1   cd $REPO && docker compose exec -T api python scripts/ingest_wikimedia_commons.py

# ── P1C Parlement suisse — mensuel.
0 3 1 * *   cd $REPO && docker compose exec -T api python scripts/ingest_parlament_ch.py

# ── P2A GDELT (couverture médiatique) — hebdomadaire, top 100 entités.
0 6 * * 1   cd $REPO && docker compose exec -T api python scripts/ingest_gdelt.py --limit 100
